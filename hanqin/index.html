suggest.sql · 建议稿，未经审核不得执行
-- 退保功能测试 · 目标保单（AI 建议稿 v3）
SELECT p.policy_no, p.holder_name, -- ⚠ 结果需脱敏
       p.effective_date, p.status
FROM   life_policy p
JOIN   rider r ON r.policy_no = p.policy_no
WHERE  p.status = 'ACTIVE'            -- 已生效
  AND  p.effective_date <= ADD_MONTHS(SYSDATE, -24)
  AND  r.rider_type IS NOT NULL       -- 有附加险
  AND  NOT EXISTS (SELECT 1 FROM claim c
                 WHERE c.policy_no = p.policy_no)  -- 未理赔
  AND  ROWNUM <= 10;
